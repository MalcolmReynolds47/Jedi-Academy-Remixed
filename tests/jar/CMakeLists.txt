function(jar_test_binary name source)
    add_executable(${name} ${source} test_support.cpp ${SharedDir}/qcommon/q_string.c ${ARGN})
    target_include_directories(${name} PRIVATE ${SPDir} ${SharedDir}
        ${CMAKE_BINARY_DIR}/shared ${GSLIncludeDirectory} ${SPDir}/rd-common ${OpenJKLibDir})
    target_compile_definitions(${name} PRIVATE ${SharedDefines} _JK2EXE)
    if(MSVC)
        target_compile_options(${name} PRIVATE /Gy /Gw /EHsc /UNDEBUG)
        target_link_options(${name} PRIVATE /OPT:REF /OPT:ICF /INCREMENTAL:NO)
    else()
        target_compile_options(${name} PRIVATE -ffunction-sections -fdata-sections -UNDEBUG)
        target_link_options(${name} PRIVATE -Wl,--gc-sections)
    endif()
endfunction()

function(jar_test name source)

    add_test(NAME ${name} COMMAND ${name})
